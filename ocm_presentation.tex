\begin{frame}{The Problem — Two Datasets, Three Systematic Differences}
  \framesubtitle{Chapter 1}

  \begin{columns}[T]

    \begin{column}{0.45\textwidth}
      \begin{block}{Our Lab Data}
        \begin{tabular}{ll}
          \toprule
          \textbf{Samples}     & 89{,}074 \\
          \textbf{Year}        & 2025 \\
          \textbf{Preparation} & Impregnation only \\
          \textbf{Mean Y(C2)}  & 5.25\% \\
          \textbf{Conditions}  & Fixed, one lab \\
          \bottomrule
        \end{tabular}
      \end{block}
    \end{column}

    \begin{column}{0.45\textwidth}
      \begin{block}{Published Literature ($\leq$2019)}
        \begin{tabular}{ll}
          \toprule
          \textbf{Samples}     & 3{,}852 \\
          \textbf{Year}        & 1982–2019 \\
          \textbf{Preparation} & 15+ methods \\
          \textbf{Mean Y(C2)}  & 8.67\% \\
          \textbf{Conditions}  & 40 years of diverse labs \\
          \bottomrule
        \end{tabular}
      \end{block}
    \end{column}

  \end{columns}

  \vspace{5pt}
  \textbf{Central question:} Can we use literature to improve our model
  \emph{without hurting it}? Three separate problems make this non-trivial:

  \vspace{3pt}
  \small
  \begin{itemize}
    \item \badge{red}{Label shift} \ Literature mean is 3.42 pp higher
      ($t = -32.2$, $p < 10^{-200}$) — systematic offset from publication bias
      and optimised operating conditions.
    \item \badge{red}{Covariate shift} \ 78.5\% of literature samples
      describe chemistry our lab has never tested.
    \item \badge{red}{Publication bias} \ Literature is \emph{skewed}, not
      just shifted — failed experiments are rarely reported, distorting the shape.
  \end{itemize}

  \sidenote{Why ML? The catalyst search space (65 elements $\times$ varying
    loadings $\times$ temperature) is too large to screen physically.
    A surrogate model guides experiments toward high-yield compositions.}

\end{frame}

% ── Slide 5 — Chapter 3: How We Measure Success ───────────────────────────────
\begin{frame}{How We Measure Success — Asymmetric 5-Fold CV}
  \framesubtitle{Chapter 3}

  \begin{columns}[T]

    \begin{column}{0.55\textwidth}
      \vspace{4pt}
      \textbf{Validation rule}
      \begin{itemize}
        \item Our 89,074 samples split into 5 folds
        \item Training fold = 4/5 of \emph{our} data $+$ any literature
        \item \alert{Validation fold = 1/5 of \emph{our} data only}
        \item Literature \emph{never} appears in validation
      \end{itemize}

      \vspace{6pt}
      \textbf{Why asymmetric?}
      \begin{itemize}
        \item The question: ``how well does the model predict
          \emph{our} experiments?''
        \item Mixing literature into validation would dilute the answer
        \item Same CV harness for all 5 methods — only training data changes
      \end{itemize}
    \end{column}

    \begin{column}{0.42\textwidth}
      \begin{block}{Metric}
        \textbf{RMSE} in units of percentage points of C$_2$ yield.\\[4pt]
        \footnotesize An RMSE of 2.1 means typical error $\approx$ 2 pp of yield.
      \end{block}

      \vspace{6pt}
      \begin{tikzpicture}
        \node[draw=navyblue, rounded corners=3pt, fill=lightgray,
              text width=0.9\linewidth, align=center, font=\small]
          {\textbf{Fold $k$:} \\
           train on $\{\text{4 our folds}\} + L_{\mathrm{extra}}$
           $\rightarrow$ predict on $\text{our fold }k$};
      \end{tikzpicture}
    \end{column}

  \end{columns}

\end{frame}

      \end{block}

      \vspace{4pt}
      \sidenote{The 3.42 pp offset is a \emph{systematic signal} the model
        cannot account for and incorrectly tries to fit.
        $\Rightarrow$ \textbf{We need to be much more selective.}}
    \end{column}

% ── Slide 7 — Chapter 5: DRST ─────────────────────────────────────────────────
\begin{frame}{DRST — Filtering by Chemical Similarity}
  \framesubtitle{Chapter 5}

  \begin{columns}[T]

    \begin{column}{0.60\textwidth}
      \includegraphics[width=\linewidth]{fig_drst_scores.png}

      \vspace{3pt}
      \footnotesize
      Histogram of $P(\text{ours} \mid x)$ for every literature sample.
      Dashed lines = candidate thresholds.
    \end{column}

    \begin{column}{0.36\textwidth}
      \textbf{How it works}
      \begin{enumerate}\footnotesize
        \item Train logistic classifier: our $=1$, lit $=0$
        \item Score each lit sample: $P(\text{ours} \mid x)$
        \item Keep only samples scoring $\geq \tau$
      \end{enumerate}

      \vspace{4pt}
      \textbf{Threshold sweep}
      \footnotesize
      \begin{tabular}{lcr}
        \toprule
        $\tau$ & kept & RMSE \\
        \midrule
        0.10 & 1168 & 2.068 \\
        0.20 &  987 & 2.034 \\
        \rowcolor{green!15}
        \textbf{0.30} & \textbf{782} & \textbf{2.019} \\
        0.40 &  543 & 2.045 \\
        \bottomrule
      \end{tabular}

      \vspace{4pt}
      \sidenote{\textbf{Unsolved:} hard cutoff is arbitrary —
        0.29 fully discarded; 0.31 gets full weight.}
    \end{column}

  \end{columns}

\end{frame}

% ── Slide 8 — Chapter 6: KMM ──────────────────────────────────────────────────
\begin{frame}{KMM — Soft Weights Instead of a Hard Filter}
  \framesubtitle{Chapter 6}

  \begin{columns}[T]

    \begin{column}{0.65\textwidth}
      \includegraphics[width=\linewidth]{fig_kmm_weights.png}
    \end{column}

    \begin{column}{0.32\textwidth}
      \textbf{How it works}
      \begin{itemize}\footnotesize
        \item Each lit sample gets weight $w_i \in [0, 10]$
        \item RBF kernel: $K(x,x')=\exp(-\|x-x'\|^2/2\sigma^2)$
        \item Solve QP: $\min \tfrac{1}{2} w^\top K_{ss} w - \kappa^\top w$
        \item $\sigma$ via median heuristic (no tuning)
      \end{itemize}

      \vspace{4pt}
      \textbf{Result}
      \footnotesize
      \begin{tabular}{lr}
        \toprule
        KMM RMSE & \textbf{2.035} \\
        vs baseline & $-4.6\%$ \\
        Near-zero weights & 78.5\% \\
        Agreement w/ DRST & $r = 0.79$ \\
        \bottomrule
      \end{tabular}

      \vspace{4pt}
      \sidenote{Two unrelated methods flag the same 78.5\% —
        real signal. Both still use labels; offset bleeds in.}
    \end{column}

  \end{columns}

\end{frame}

% ── Slide 13 — Chapter 9a: Critical Analysis ──────────────────────────────────
\begin{frame}[allowframebreaks]{Critical Analysis — Per-Method Review}
  \framesubtitle{Chapter 9}

  \begin{columns}[T]

    \begin{column}{0.52\textwidth}
      \textbf{Honest critique of each method}
      \vspace{4pt}
      \scriptsize
      \begin{tabular}{@{}p{1.6cm}p{5.0cm}@{}}
        \toprule
        \textbf{Method} & \textbf{Main limitation} \\
        \midrule
        Naive merge & Negative control — makes the model worse \\[2pt]
        DRST & Hard cliff; discards $\sim$80\%; $\tau$ tuned on the reported CV \\[2pt]
        KMM & $O(n^2)$ kernel; $\sigma$-sensitive; labels still in loss; ties DRST \\[2pt]
        \rowcolor{green!12}
        Prior FT & Best — Stage-1 on 782 rows $\to$ high variance; two models \\
        \bottomrule
      \end{tabular}
    \end{column}

    \begin{column}{0.44\textwidth}
      \textbf{Cross-cutting limits}
      \vspace{4pt}
      \begin{itemize}\footnotesize
        \item All methods evaluated on our-data CV only; OOD tested for Prior FT alone
        \item Point predictions — no uncertainty estimates to guide experiment selection
        \item Y$>$15\% ceiling is \emph{data}-bound (missing GHSV, CH$_4$:O$_2$,
          pressure), not model-bound
      \end{itemize}

      \vspace{6pt}
      \sidenote{\textbf{Key insight:} the ceiling effect and much of the label shift
        share the same root cause — missing operating-condition features.
        Fix the data; the models follow.}
    \end{column}

  \end{columns}

\end{frame}

% ── Slide 14 — Chapter 9b: What's Next ───────────────────────────────────────
\begin{frame}[allowframebreaks]{What's Next — Priority-Ordered}
  \framesubtitle{Chapter 9}

  \begin{columns}[T]

    \begin{column}{0.48\textwidth}
      \begin{enumerate}\footnotesize
        \item \badge{green}{Highest impact} \textbf{Add GHSV, CH$_4$:O$_2$,
          pressure} — fixes the Y$>$15\% ceiling and likely resolves much of
          the label shift too.
        \vspace{3pt}
        \item \textbf{Bag the Stage-1 expert} — ensemble of XGBoost models
          reduces high variance from only 782 training samples.
        \vspace{3pt}
        \item \textbf{Add uncertainty} (conformal prediction or quantile
          regression) — flag low-confidence picks for the experiment loop.
      \end{enumerate}
    \end{column}

    \begin{column}{0.48\textwidth}
      \begin{enumerate}\footnotesize
        \setcounter{enumi}{3}
        \item \textbf{Stack DRST + KMM} inside the Prior FT pipeline rather
          than running them as three separate methods.
        \vspace{3pt}
        \item \textbf{Neural domain adaptation} (DANN, CORAL) once the feature
          set is richer — currently under-powered for deep methods.
        \vspace{3pt}
        \item \textbf{Close the active-learning loop} — model proposes
          candidates, lab measures them, model retrains, reducing label shift
          over time.
      \end{enumerate}
    \end{column}

  \end{columns}

\end{frame}

          Ba, Mn, La, Ce — and the literature prior — drive predictions.
      \end{itemize}

      to the feature set — the high-yield ceiling is data-bound,
      not model-bound.
    };
