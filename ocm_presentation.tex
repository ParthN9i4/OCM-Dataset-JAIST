% ── Slide 2 — Chapter 1: The Problem ──────────────────────────────────────────
\begin{frame}{The Problem — Two Datasets, Three Systematic Differences}

  \begin{columns}[T]

    \begin{column}{0.45\textwidth}
      \begin{block}{Our Lab Data}
        \begin{tabular}{ll}
          \toprule
          \textbf{Samples}     & 89{,}074 \\
          \textbf{Year}        & 2025 \\
          \textbf{Preparation} & Impregnation only \\
          \textbf{Mean Y(C2)}  & 5.25\% \\
          \textbf{Conditions}  & Fixed, one lab \\
          \bottomrule
        \end{tabular}
      \end{block}
    \end{column}

    \begin{column}{0.45\textwidth}
      \begin{block}{Published Literature ($\leq$2019)}
        \begin{tabular}{ll}
          \toprule
          \textbf{Samples}     & 3{,}852 \\
          \textbf{Year}        & 1982–2019 \\
          \textbf{Preparation} & 15+ methods \\
          \textbf{Mean Y(C2)}  & 8.67\% \\
          \textbf{Conditions}  & 40 years of diverse labs \\
          \bottomrule
        \end{tabular}
      \end{block}
    \end{column}

  \end{columns}

  \vspace{6pt}
  \textbf{Central question:} Can we use literature to improve our model
  \emph{without hurting it}? Three separate problems make this non-trivial:

  \vspace{4pt}
  \small
  \begin{itemize}
    \item \badge{red}{1.\ Label shift} \ Literature mean is 3.42 pp higher
      ($t = -32.2$, $p < 10^{-200}$) — a systematic offset from publication
      bias and optimised operating conditions.
    \item \badge{red}{2.\ Covariate shift} \ 78.5\% of literature samples
      describe chemistry our lab has never tested.
    \item \badge{red}{3.\ Publication bias} \ Literature is \emph{skewed}, not
      just shifted — failed experiments are rarely reported, over-weighting
      high yields and distorting the distribution shape.
  \end{itemize}

  \vspace{4pt}
  \colorbox{lightgray}{\parbox{0.93\linewidth}{\scriptsize
    \textbf{Why ML?} The catalyst search space (65 elements $\times$ varying
    loadings $\times$ temperature) is too large to screen physically.
    A surrogate model guides experiments toward high-yield compositions.
  }}

\end{frame}

% ── Slide 4 — Chapter 3: How We Measure Success ───────────────────────────────
\begin{frame}{How We Measure Success — Asymmetric 5-Fold CV}

  \begin{columns}[T]

    \begin{column}{0.55\textwidth}
      \vspace{6pt}
      \textbf{Validation rule}
      \begin{itemize}
        \item Our 89,074 samples split into 5 folds
        \item Training fold = 4/5 of \emph{our} data $+$ any literature
        \item \alert{Validation fold = 1/5 of \emph{our} data only}
        \item Literature \emph{never} appears in validation
      \end{itemize}

      \vspace{8pt}
      \textbf{Why asymmetric?}
      \begin{itemize}
        \item The question is: ``how well does the model predict
          \emph{our} experiments?''
        \item Mixing literature into validation would dilute the answer
        \item Same CV harness is used for all 5 methods — only the
          training data changes
      \end{itemize}
    \end{column}

    \begin{column}{0.42\textwidth}
      \begin{block}{Metric}
        \textbf{RMSE} in units of percentage points of C$_2$ yield.\\[4pt]
        \footnotesize An RMSE of 2.1 means typical error $\approx$ 2 pp of yield.
      \end{block}

      \vspace{6pt}
      \begin{tikzpicture}
        \node[draw=navyblue, rounded corners=3pt, fill=lightgray,
              text width=0.9\linewidth, align=center, font=\small]
          {\textbf{Fold $k$:} \\
           train on $\{\text{4 our folds}\} + L_{\mathrm{extra}}$
           $\rightarrow$ predict on $\text{our fold }k$};
      \end{tikzpicture}
    \end{column}

  \end{columns}

\end{frame}

      \begin{tabular}{lr}

        Baseline (our data only) & \textbf{2.133} \\
        \bottomrule
      \end{tabular}

      \vspace{14pt}

      \begin{tabular}{lr}
        \toprule

        \midrule
        Baseline & 2.133 \\

        \bottomrule
      \end{tabular}
      \vspace{4pt}\\

    \end{column}

    \begin{column}{0.48\textwidth}
      \begin{block}{Why did it fail?}
        \small
        Model sees two catalysts with nearly identical composition:
        \begin{itemize}\small
          \item One labelled \textbf{5.2\%} (our lab)
          \item One labelled \textbf{8.7\%} (literature)
        \end{itemize}
        No feature explains the gap — flow rate, gas ratio, pressure
        are absent. The model hedges and predicts both badly.
      \end{block}

      \vspace{6pt}
      \colorbox{lightgray}{\parbox{\linewidth}{\small
        The 3.42 pp offset is a \emph{systematic signal} the model
        cannot account for and incorrectly tries to fit. Mixing the
        two datasets can perform \emph{worse} than using ours alone.\\[4pt]
        $\Rightarrow$ \textbf{We need to be much more selective.}
      }}
    \end{column}

  \end{columns}

\end{frame}

% ── Slide 6 — Chapter 5: DRST ─────────────────────────────────────────────────
\begin{frame}{DRST — Filtering by Chemical Similarity}

  \begin{columns}[T]

    \begin{column}{0.52\textwidth}
      \includegraphics[width=\linewidth]{fig_drst_scores.png}

      \vspace{4pt}
      \footnotesize
      Histogram of $P(\text{ours} \mid x)$ for every literature sample.
      Dashed lines = candidate thresholds.
    \end{column}

    \begin{column}{0.44\textwidth}
      \textbf{How it works}
      \begin{enumerate}\small
        \item Train \textit{logistic classifier}
          (finds a boundary in element-space):\\
          our $=1$, literature $=0$
        \item Score each lit sample: $P(\text{ours} \mid x)$
        \item Keep only samples scoring $\geq \tau$
      \end{enumerate}

      \vspace{4pt}
      \textbf{Threshold sweep} (Stone 1974 CV selection)
      \begin{tabular}{lcr}
        \toprule
        $\tau$ & kept & CV RMSE \\
        \midrule
        0.10 & 1168 & 2.068 \\
        0.20 &  987 & 2.034 \\
        \rowcolor{green!15}
        \textbf{0.30} & \textbf{782} & \textbf{2.019} \\
        0.40 &  543 & 2.045 \\
        \bottomrule
      \end{tabular}

      \vspace{4pt}
      \colorbox{lightgray}{\parbox{\linewidth}{\scriptsize
        \textbf{Unsolved:} the hard cutoff is arbitrary — a sample at
        0.29 is fully discarded; 0.31 gets full weight.
      }}
    \end{column}

  \end{columns}

\end{frame}

% ── Slide 7 — Chapter 6: KMM ──────────────────────────────────────────────────
\begin{frame}{KMM — Soft Weights Instead of a Hard Filter}

  \begin{columns}[T]

    \begin{column}{0.55\textwidth}
      \includegraphics[width=\linewidth]{fig_kmm_weights.png}
    \end{column}

    \begin{column}{0.42\textwidth}
      \textbf{How it works}
      \begin{itemize}\small
        \item Each lit sample gets a weight $w_i \in [0, 10]$
        \item \textit{RBF kernel} measures chemical similarity:
          $K(x,x')=\exp\!\bigl(-\|x-x'\|^2/2\sigma^2\bigr)$
        \item Solve QP: $\min \frac{1}{2} w^\top K_{ss} w - \kappa^\top w$
        \item $\sigma$ via median heuristic (data-driven, no tuning)
      \end{itemize}

      \vspace{4pt}
      \textbf{Result}
      \begin{tabular}{lr}
        \toprule
        Metric & Value \\
        \midrule
        KMM CV RMSE & \textbf{2.035} \\
        $\Delta$ vs baseline & \textcolor{darkgray}{$-4.6\%$} \\
        Near-zero weights & 78.5\% \\
        Agreement w/ DRST & \textbf{$r = 0.79$} \\
        \bottomrule
      \end{tabular}

      \vspace{4pt}
      \colorbox{lightgray}{\parbox{\linewidth}{\scriptsize
        \textbf{Reassuring:} two unrelated methods flag the same 78.5\%
        of samples $\Rightarrow$ real signal, not artifact.\\[2pt]
        \textbf{Still unsolved:} literature labels remain in training,
        so the 3.42 pp offset still bleeds in.
      }}
    \end{column}

  \end{columns}

\end{frame}

\begin{frame}{Prior Feature Transfer — The Winning Pipeline}

  \begin{tikzpicture}[
      box/.style={rectangle, rounded corners=4pt, draw=navyblue, fill=lightgray,
                  text width=2.6cm, align=center, minimum height=1.0cm, font=\small},
      arrow/.style={-{Stealth}, thick, color=accent},
      label/.style={font=\scriptsize, color=darkgray},
      every node/.style={font=\small}
    ]

    \node[box, fill=accent!15] (lit) {Literature\\3{,}852 samples\\(DRST filtered)};
    \node[box, right=1.4cm of lit, fill=accent!25] (xgb) {Stage 1\\XGBoost\\pre-train};
    \node[box, right=1.4cm of xgb, fill=amber!20] (prior) {\texttt{lit\_prior}\\prediction\\(68th feature)};

    \draw[arrow] (lit) -- (xgb);
    \draw[arrow] (xgb) -- (prior);

    \node[box, below=1.4cm of lit, fill=green!15] (lab) {Our Lab Data\\89{,}074 samples};
    \node[box, right=1.4cm of lab, fill=navyblue!10] (merge) {Append\\68\textsuperscript{th} feature};
    \node[box, right=1.4cm of merge, fill=green!25] (lgbm) {Stage 2\\LightGBM\\\emph{our labels only}};
    \node[box, right=1.4cm of lgbm, fill=navyblue!20] (out) {Y(C$_2$)\%\\prediction};

    \draw[arrow] (lab) -- (merge);
    \draw[arrow] (merge) -- (lgbm);
    \draw[arrow] (lgbm) -- (out);

    \draw[arrow, color=amber] (prior) -- (merge);

    \node[label, above=0.2cm of xgb] {Stage 1 — literature expert};
    \node[label, above=0.2cm of lgbm] {Stage 2 — fine-tune on our labels};

  \end{tikzpicture}

  \vspace{6pt}
  \small
  \textbf{Key property:} Stage 2 \emph{never sees literature labels}.
  The 3.42 pp offset affects the \emph{value} of the 68th feature —
  which Stage 2 can calibrate (``when the expert says 8, my lab gets $\sim$4.5'') —
  not the training loss, which it could not fix. A corrupted label can't be
  un-corrupted; a feature value is just another thing to learn around.
  Stage 1 uses XGBoost (conservative on 782 samples); Stage 2 uses LightGBM
  (fast on 89k rows).

\end{frame}

% ── Slide 11 — Critical Analysis & Next Steps ─────────────────────────────────
\begin{frame}{Critical Analysis \& Next Steps}

  \begin{columns}[T]

    \begin{column}{0.55\textwidth}
      \textbf{Honest critique of each method}
      \vspace{2pt}
      \footnotesize
      \begin{tabular}{@{}p{2.0cm}p{4.3cm}@{}}
        \toprule
        \textbf{Method} & \textbf{Main limitation} \\
        \midrule
        Naive merge & Negative control only — harms the model \\
        DRST & Hard cliff; discards $\sim$80\% of lit; $\tau$ tuned on the reported CV \\
        KMM & Slow $n{\times}n$ kernel; $\sigma$-sensitive; still uses labels; ties DRST \\
        \rowcolor{green!12}
        Prior FT & Best, but Stage-1 on only 782 rows $\rightarrow$ high variance; 2 models \\
        \bottomrule
      \end{tabular}

      \vspace{6pt}
      \textbf{Cross-cutting limits}
      \begin{itemize}\footnotesize
        \item Scored on our-data CV only; OOD checked for Prior FT alone
        \item Point predictions — no uncertainty for picking experiments
        \item Y$>$15\% ceiling is \emph{data}-bound, not model-bound
      \end{itemize}
    \end{column}

    \begin{column}{0.42\textwidth}
      \textbf{What's next (priority-ordered)}
      \vspace{2pt}
      \begin{enumerate}\small
        \item \badge{green}{Highest} Add GHSV, CH$_4$:O$_2$, pressure —
          fixes the ceiling and likely the label shift
        \item Bag the Stage-1 expert to cut its variance
        \item Add uncertainty (quantile / conformal) for experiment selection
        \item Stack DRST + KMM into the Prior FT pipeline
        \item Neural domain adaptation (DANN / CORAL) once features are richer
        \item Close the active-learning loop: propose $\to$ measure $\to$ retrain
      \end{enumerate}
    \end{column}

  \end{columns}

\end{frame}
